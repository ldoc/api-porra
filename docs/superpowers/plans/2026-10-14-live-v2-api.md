# Live v2 — Backend (`api-porra`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Persistir un log de eventos por partido (colección `liveevents`) derivado de Sofascore y exponerlo por API para que el front reconstruya el live.

**Architecture:** Scraper local (`scripts/scrapeLiveMatches.js`) obtiene de Sofascore una *observación* normalizada (`scrapMatchSource`) y deriva eventos nuevos con una función pura (`api/liveEvents.js:deriveEvents`), insertándolos de forma idempotente en Mongo (`liveevents`, TTL fin de día). El servidor expone `GET /api/live/events` (full) y `?since=` (delta). Se conserva la escritura de snapshots `livematches` como respaldo.

**Tech Stack:** Node.js ESM, `http` nativo, Mongoose (MongoDB Atlas), `node:test`, `node-curl-impersonate`.

## Global Constraints

- ES Modules (`import`/`export`), sin CommonJS.
- Nombres de funciones/variables en español (patrón del repo).
- Cuerpos de respuesta 200 **deterministas** (ETag débil SHA-1; nada de timestamps por petición salvo cursores derivados de datos).
- Endpoints GET pasan por `checkPhaseConsistency(req, res)`; responden 409 `PHASE_CHANGED` si el cliente desfase.
- Sin auth para `/api/live/events` (igual que `/api/live`); el DELETE admin requiere `verifyAdmin(req)`.
- Mongo: colección explícita `liveevents`; TTL `{ expireAt: 1 }, { expireAfterSeconds: 0 }`; índice único `{ eventId: 1, sourceId: 1 }`; índice `{ ts: 1 }`.
- `expireAt` = 23:59:59 UTC del día del evento.
- Tests con `node:test` (`import test from 'node:test'; import assert from 'node:assert/strict';`).
- Toda feature nueva que calcule puntos debe respetar el perfil invitado (`isGuest`) — aquí no aplica (no calcula puntos en backend), pero no exponer datos de usuarios.

```js
// api/liveEvents.js
export const EVENT_TYPES = [
  'lineup', 'status', 'goal', 'goal_cancelled', 'penalty_saved',
  'penalty_missed', 'card', 'substitution', 'rating', 'score', 'match_final'
];

export function expireAtEndOfDayUTC(nowMs) {
  const d = new Date(nowMs);
  return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate(), 23, 59, 59, 0));
}

export function foldScore(events) {
  const byId = new Map(events.map(e => [e.sourceId, e]));
  let home = 0, away = 0;
  for (const e of events) {
    if (e.type === 'goal') {
      if (e.payload?.isHome) home++; else away++;
    } else if (e.type === 'score') {
      home = e.payload?.scoreHome ?? home;
      away = e.payload?.scoreAway ?? away;
    } else if (e.type === 'goal_cancelled') {
      const g = byId.get(e.payload?.ofSourceId);
      if (g) { if (g.payload?.isHome) home--; else away--; }
    }
  }
  return { home: Math.max(0, home), away: Math.max(0, away) };
}

export function deriveEvents(obs, existing = [], nowMs = Date.now()) {
  const have = new Set(existing.map(e => e.sourceId));
  const out = [];
  const expireAt = expireAtEndOfDayUTC(nowMs);
  const push = (ev) => {
    if (have.has(ev.sourceId)) return;
    have.add(ev.sourceId);
    out.push({
      eventId: obs.eventId, ts: new Date(nowMs), second: 0,
      minute: ev.minute ?? null, type: ev.type,
      teamId: ev.teamId ?? null, playerId: ev.playerId ?? null,
      sourceId: ev.sourceId, payload: ev.payload ?? {}, expireAt
    });
  };

  if (Array.isArray(obs.starters) && obs.starters.length) {
    push({ sourceId: 'lineup|' + [...obs.starters].sort((a, b) => a - b).join(','), type: 'lineup', minute: 0, payload: { starters: obs.starters } });
  }

  for (const inc of obs.incidents || []) {
    if (inc.type === 'goal') {
      push({ sourceId: 'goal|' + inc.sourceId, type: 'goal', teamId: inc.teamId, playerId: inc.playerId, minute: inc.time ?? null,
        payload: { scorerId: inc.playerId, assistId: inc.assistId ?? null, isPenalty: inc.class === 'penalty', isOwnGoal: inc.class === 'ownGoal', isHome: !!inc.isHome, scoreHome: inc.homeScore ?? null, scoreAway: inc.awayScore ?? null } });
    }
  }

  if (obs.status) {
    push({ sourceId: 'status|' + obs.status, type: 'status', minute: obs.minute ?? null, payload: { status: obs.status, period: obs.period ?? null } });
  }

  for (const [pid, rating] of Object.entries(obs.ratings || {})) {
    push({ sourceId: `rating|${pid}|${rating}`, type: 'rating', playerId: Number(pid), minute: obs.minute ?? null, payload: { rating } });
  }

  const folded = foldScore([...existing, ...out]);
  if (obs.score && (folded.home !== obs.score.home || folded.away !== obs.score.away)) {
    push({ sourceId: `score|${obs.score.home}-${obs.score.away}`, type: 'score', minute: obs.minute ?? null, payload: { scoreHome: obs.score.home, scoreAway: obs.score.away } });
  }

  return out;
}
```

```js
export function normalizeIncidents(incidentsData, homeTeamId, awayTeamId) {
  const out = [];
  for (const inc of incidentsData?.incidents || []) {
    const teamId = inc.isHome ? homeTeamId : awayTeamId;
    const base = { sourceId: String(inc.id), time: inc.time ?? null, addedTime: inc.addedTime ?? null, teamId };
    switch (inc.incidentType) {
      case 'goal':
        out.push({ ...base, type: 'goal', class: inc.incidentClass,
          playerId: inc.player?.id ?? null,
          assistId: inc.footballPassingNetworkAction?.[0]?.player?.id ?? null,
          homeScore: inc.homeScore ?? null, awayScore: inc.awayScore ?? null, isHome: !!inc.isHome });
        break;
      case 'inGamePenalty':
        if (inc.incidentClass === 'missed') {
          out.push({ ...base, type: 'penalty_saved', keeperId: inc.footballPassingNetworkAction?.[0]?.goalkeeper?.id ?? null, takerId: inc.player?.id ?? null });
          out.push({ ...base, type: 'penalty_missed', takerId: inc.player?.id ?? null, sourceId: String(inc.id) + '-taker' });
        }
        break;
      case 'card':
        out.push({ ...base, type: 'card', class: inc.incidentClass, cardType: inc.incidentClass, playerId: inc.player?.id ?? null });
        break;
      case 'substitution':
        out.push({ ...base, type: 'substitution', inId: inc.playerIn?.id ?? null, outId: inc.playerOut?.id ?? null });
        break;
      case 'varDecision':
        if (inc.incidentClass === 'goalDisallowed') {
          out.push({ ...base, type: 'goal_cancelled', class: inc.incidentClass, playerId: inc.player?.id ?? null });
        }
        break;
      default:
        break;
    }
  }
  return out;
}
```

```js
export async function scrapMatchSource(eventId) {
  const eventUrl = `https://www.sofascore.com/api/v1/event/${eventId}`;
  const lineupsUrl = `https://www.sofascore.com/api/v1/event/${eventId}/lineups`;
  const incidentsUrl = `https://www.sofascore.com/api/v1/event/${eventId}/incidents`;
  const fantasyUrl = `https://www.sofascore.com/api/v1/fantasy/event/${eventId}`;

  const [eventData, incidentsData, fantasyData] = await Promise.all([
    fetchSofascore(eventUrl),
    fetchSofascore(incidentsUrl).catch(() => null),
    fetchSofascore(fantasyUrl).catch(() => null)
  ]);
  const ev = eventData.event;
  const homeTeamId = ev.homeTeam?.id;
  const awayTeamId = ev.awayTeam?.id;

  let lineupsData = null;
  try { lineupsData = await fetchSofascore(lineupsUrl); } catch (e) {
    const notStarted = ev.status?.type === 'notstarted' || ev.status?.type === 'willsoonstart';
    if (!notStarted) throw e;
  }

  const players = [...(lineupsData?.home?.players || []), ...(lineupsData?.away?.players || [])];
  const starters = players.filter(p => !p.substitute).map(p => p.player?.id).filter(Boolean);
  const ratings = {};
  const minutes = {};
  for (const p of players) {
    const id = p.player?.id;
    if (!id) continue;
    if (p.statistics?.rating != null) ratings[id] = p.statistics.rating;
    if (p.statistics?.minutesPlayed != null) minutes[id] = p.statistics.minutesPlayed;
  }

  return {
    eventId,
    status: null, // lo resuelve el scraper con liveStatusFromSofascore
    statusType: ev.status?.type ?? null,
    statusCode: ev.status?.code ?? null,
    minute: liveMinute(ev.lastPeriod, ev.time?.currentPeriodStartTimestamp, Math.floor(Date.now() / 1000)),
    period: ev.lastPeriod ?? null,
    score: { home: ev.homeScore?.display ?? 0, away: ev.awayScore?.display ?? 0 },
    homeTeamId, awayTeamId,
    starters, ratings, minutes,
    incidents: normalizeIncidents(incidentsData, homeTeamId, awayTeamId)
  };
}
```

```js
// api/live.js — añadir al final
export function buildLiveEventsResponse(docs) {
  const events = [...(docs || [])].sort((a, b) => new Date(a.ts) - new Date(b.ts));
  const max = events.reduce((acc, d) => Math.max(acc, new Date(d.ts).getTime()), 0);
  return { ok: true, events, serverTime: max ? new Date(max).toISOString() : null };
}
```
